**目标：** 在不改动信号管道的前提下，优先解决短时 lock/unlock 抖动，并把事件日志从“已尝试”改为“已确认结果”。

**技术栈：** Swift, XCTest, macOS AppKit, os.Logger, CSV logging

```swift
import Foundation

final class ScriptRunner {
    static let shared = ScriptRunner()

    enum EventSource: Equatable {
        case auto
        case user
        case system
    }

    enum EventResult: String, Equatable {
        case success
        case fail
        case timeout
    }

    private let dedupWindow: TimeInterval
    private let nowProvider: () -> Date
    private var lastEventTimestamps: [String: Date] = [:]

    init(dedupWindow: TimeInterval = 3.0,
         nowProvider: @escaping () -> Date = Date.init) {
        self.dedupWindow = dedupWindow
        self.nowProvider = nowProvider
    }

    private init() {
        self.dedupWindow = 3.0
        self.nowProvider = Date.init
    }

    // MARK: - Event Logging

    @discardableResult
    func logEventIfNeeded(
        _ event: String,
        rssi: Int? = nil,
        result: EventResult? = nil,
        latencyMs: Int? = nil,
        source: EventSource? = nil,
        effectiveRSSI: Double? = nil,
        device: String? = nil
    ) -> Bool {
        let now = nowProvider()
        if let last = lastEventTimestamps[event],
           now.timeIntervalSince(last) < dedupWindow {
            return false
        }

        lastEventTimestamps[event] = now
        let line = buildEventLine(
            event,
            rssi: rssi,
            result: result,
            latencyMs: latencyMs,
            source: source,
            effectiveRSSI: effectiveRSSI,
            device: device
        )
        appendLogLine(line)
        return true
    }

    func buildEventLine(
        _ event: String,
        rssi: Int?,
        result: EventResult?,
        latencyMs: Int?,
        source: EventSource?,
        effectiveRSSI: Double?,
        device: String?
    ) -> String {
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyy-MM-dd HH:mm:ss"
        let ts = formatter.string(from: nowProvider())
        let rssiStr = rssi.map { String($0) } ?? "N/A"
        var components = [
            "\(ts) | \(event) | RSSI: \(rssiStr)",
            "result=\(result?.rawValue ?? "N/A")",
            "latencyMs=\(latencyMs.map(String.init) ?? "N/A")",
            "source=\(source.map { String(describing: $0) } ?? "N/A")",
            "effectiveRSSI=\(effectiveRSSI.map { String(format: "%.2f", $0) } ?? "N/A")",
            "device=\(device ?? "N/A")"
        ]
        return components.joined(separator: " | ")
    }

    // MARK: - User Script Execution

    func runScript(_ arg: String, rssi: Int? = nil, deviceName: String? = nil, uuid: UUID? = nil) {
        guard let directory = try? FileManager.default.url(for: .applicationScriptsDirectory, in: .userDomainMask, appropriateFor: nil, create: true) else { return }
        let file = directory.appendingPathComponent("event")
        let process = Process()
        process.executableURL = file
        var args = [arg]
        if let r = rssi { args.append(String(r)) }
        if let name = deviceName { args.append(name) }
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyy-MM-dd'T'HH:mm:ss"
        args.append(formatter.string(from: Date()))
        process.arguments = args
        try? process.run()
    }

    // MARK: - Legacy compatible entry

    func logEvent(_ event: String, rssi: Int? = nil) {
        logEventIfNeeded(event, rssi: rssi, result: nil, latencyMs: nil, source: nil, effectiveRSSI: nil, device: nil)
    }

    private func appendLogLine(_ line: String) {
        guard let dir = try? FileManager.default.url(for: .applicationSupportDirectory, in: .userDomainMask, appropriateFor: nil, create: true) else { return }
        let logDir = dir.appendingPathComponent("FUnlock", isDirectory: true)
        try? FileManager.default.createDirectory(at: logDir, withIntermediateDirectories: true)
        let logFile = logDir.appendingPathComponent("events.log")
        let data = (line + "\n").data(using: .utf8) ?? Data()
        if FileManager.default.fileExists(atPath: logFile.path) {
            if let handle = try? FileHandle(forWritingTo: logFile) {
                handle.seekToEndOfFile()
                handle.write(data)
                handle.closeFile()
            }
        } else {
            try? data.write(to: logFile)
        }
    }
}
```

extension FUnManager {

    convenience init(
        unlockCooldown: TimeInterval,
        postLockSettle: TimeInterval,
        eventDedupWindow: TimeInterval,
        nowProvider: @escaping () -> Date = Date.init
    ) {
        self.init(fun: FUn(), inputMonitor: nil)
        self.unlockCooldown = unlockCooldown
        self.postLockSettle = postLockSettle
        self.eventDedupWindow = eventDedupWindow
        self.nowProvider = nowProvider
    }

    var lastAutoLockTime: Date? { _lastAutoLockTime }

    func markAutoLocked(now: Date) {
        _lastAutoLockTime = now
    }

    @discardableResult
    func tryAcceptUnlockTrigger() -> Bool {
        let now = nowProvider()

        if let lastUnlock = _lastUnlockAcceptedTime,
           now.timeIntervalSince(lastUnlock) < unlockCooldown {
            return false
        }

        if let lastLock = _lastAutoLockTime,
           now.timeIntervalSince(lastLock) < postLockSettle {
            return false
        }

        _lastUnlockAcceptedTime = now
        return true
    }
}

import Foundation

enum TelemetryEvent: String {
    case autoUnlock = "auto_unlock"
    case autoLock = "auto_lock"
    case abnormalAlert = "abnormal_alert"
}

struct TelemetryRecord {
    let timestamp: Date
    let eventType: TelemetryEvent
    let deviceModel: String?
    let rawRSSI: Int
    let kalmanRSSI: Double
    let effectiveRSSI: Double
    let slope: Double
    let isAnomalous: Bool
    let result: ScriptRunner.EventResult?
    let latencyMs: Int?
    let injectTime: Date?
    let confirmTime: Date?
}

final class TelemetryLogger {
    static let shared = TelemetryLogger(inMemory: false)

    private let maxFileSize: UInt64 = 5 * 1024 * 1024
    private let queue = DispatchQueue(label: "com.funlock.telemetry", qos: .utility)
    private let inMemory: Bool
    private var memoryLines: [String] = []

    private var logDirectory: URL {
        FileManager.default.homeDirectoryForCurrentUser.appendingPathComponent("Library/Logs/FUnlock")
    }

    private var logFile: URL {
        logDirectory.appendingPathComponent("shadow_telemetry.csv")
    }

    init(inMemory: Bool) {
        self.inMemory = inMemory
        if !inMemory {
            ensureDirectory()
        }
    }

    func log(
        event: TelemetryEvent,
        deviceModel: String?,
        rawRSSI: Int,
        kalmanRSSI: Double,
        effectiveRSSI: Double,
        slope: Double,
        isAnomalous: Bool,
        result: ScriptRunner.EventResult? = nil,
        latencyMs: Int? = nil,
        injectTime: Date? = nil,
        confirmTime: Date? = nil
    ) {
        let record = TelemetryRecord(
            timestamp: Date(),
            eventType: event,
            deviceModel: deviceModel,
            rawRSSI: rawRSSI,
            kalmanRSSI: kalmanRSSI,
            effectiveRSSI: effectiveRSSI,
            slope: slope,
            isAnomalous: isAnomalous,
            result: result,
            latencyMs: latencyMs,
            injectTime: injectTime,
            confirmTime: confirmTime
        )

        if inMemory {
            memoryLines.append(buildRow(for: record))
            return
        }

        queue.async { [weak self] in
            self?.writeRecord(record)
        }
    }

    func currentHeader() -> String {
        buildHeader()
    }

    func lastRow() -> String {
        memoryLines.last ?? ""
    }

    private func ensureDirectory() {
        try? FileManager.default.createDirectory(at: logDirectory, withIntermediateDirectories: true)
    }

    private func writeRecord(_ record: TelemetryRecord) {
        if let attrs = try? FileManager.default.attributesOfItem(atPath: logFile.path),
           let size = attrs[.size] as? UInt64,
           size > maxFileSize {
            try? FileManager.default.removeItem(at: logFile)
            writeHeaderIfNeeded()
        }

        writeHeaderIfNeeded()

        let data = (buildRow(for: record) + "\n").data(using: .utf8) ?? Data()
        if FileManager.default.fileExists(atPath: logFile.path) {
            if let handle = try? FileHandle(forWritingTo: logFile) {
                handle.seekToEndOfFile()
                handle.write(data)
                handle.closeFile()
            }
        } else {
            try? data.write(to: logFile)
        }
    }

    private func writeHeaderIfNeeded() {
        guard !FileManager.default.fileExists(atPath: logFile.path) else { return }
        try? buildHeader().data(using: .utf8)?.write(to: logFile)
    }

    private func buildHeader() -> String {
        "Timestamp,Event_Type,Device_Model,Raw_RSSI,Kalman_RSSI,Effective_RSSI,Slope,Is_Anomalous,Result,LatencyMs,InjectTime,ConfirmTime\n"
    }

    private func buildRow(for record: TelemetryRecord) -> String {
        let ts = formatTimestamp(record.timestamp)
        let model = record.deviceModel ?? "unknown"
        let inject = record.injectTime.map(formatTimestamp) ?? "N/A"
        let confirm = record.confirmTime.map(formatTimestamp) ?? "N/A"
        return [
            "\(ts),\(record.eventType.rawValue),\(model)",
            "\(record.rawRSSI)",
            String(format: "%.2f", record.kalmanRSSI),
            String(format: "%.2f", record.effectiveRSSI),
            String(format: "%.4f", record.slope),
            "\(record.isAnomalous)",
            "\(record.result?.rawValue ?? "N/A")",
            "\(record.latencyMs.map(String.init) ?? "N/A")",
            "\(inject)",
            "\(confirm)"
        ].joined(separator: ",")
    }

    private func formatTimestamp(_ date: Date) -> String {
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyy-MM-dd HH:mm:ss.SSS"
        return formatter.string(from: date)
    }
}
